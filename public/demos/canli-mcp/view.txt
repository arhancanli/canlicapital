function esc(v) {
  return String(v ?? "").replace(/[&<>"']/g, (c) => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", "\"": "&quot;", "'": "&#39;" })[c]);
}

const ISO = /^\d{4}-\d{2}-\d{2}/;
const isNum = (v) => typeof v === "number" && Number.isFinite(v);
const isUrl = (v) => /^https?:\/\//.test(String(v ?? ""));
const link = (url, html) => `<a href="${esc(url)}" target="_blank" rel="noopener noreferrer">${html}</a>`;
const words = (v) => esc(String(v ?? "").replace(/_/g, " "));
function linkText(url) {
  const m = /^https?:\/\/([^/?#]+)([^?#]*)/.exec(url), tail = (m?.[2] ?? "").split("/").filter(Boolean).pop();
  const s = tail ? `${m[1]}/…/${tail}` : m?.[1] ?? url;
  return s.length > 60 ? `${s.slice(0, 59)}…` : s;
}
const unitAfter = (u) => (u === "%" ? "%" : u ? ` <span class="sub">${esc(u)}</span>` : "");

function fmt(v, min = 0) {
  if (!isNum(v)) return esc(v ?? "");
  const abs = Math.abs(v);
  const digits = abs >= 1000 ? 0 : abs >= 1 ? 2 : 4;
  return v.toLocaleString("en-US", { maximumFractionDigits: Math.max(digits, min), minimumFractionDigits: min });
}

function short(v) {
  const a = Math.abs(v);
  for (const [d, u] of [[1e12, "T"], [1e9, "B"], [1e6, "M"]]) if (a >= d) return `${Number((v / d).toPrecision(4)).toLocaleString("en-US")}${u}`;
  return fmt(v);
}

const VERDICT = { confirmed: "ok", "confirmed as first reported": "ok", contradicted: "bad", "not reported": "warn", "not checkable": "muted" };
function badge(text) {
  const t = String(text ?? "");
  const kind = VERDICT[t] ?? (/^(above|below|within)/.test(t) ? "info" : /^not placed/.test(t) ? "muted" : null);
  return kind ? `<span class="badge ${kind}">${esc(t)}</span>` : esc(t);
}

function table({ columns, rows, title, note, maxRows = 200 }) {
  let cols = Array.isArray(columns) ? columns : [];
  let body = (Array.isArray(rows) ? rows : []).slice(0, maxRows).map((r) => (Array.isArray(r) ? r : [r]));
  let concepts = null;
  if (cols[1] === "concept") {
    concepts = body.map((r) => r[1]);
    cols = [cols[0], ...cols.slice(2)];
    body = body.map((r) => [r[0], ...r.slice(2)]);
  }
  const filled = cols.map((_, i) => i === 0 || body.some((r) => r[i] !== null && r[i] !== undefined && r[i] !== ""));
  if (filled.includes(false)) { cols = cols.filter((_, i) => filled[i]); body = body.map((r) => r.filter((_, i) => filled[i])); }
  const badgeCol = new Set(cols.map((c, i) => (/^(verdict|position)$/.test(String(c)) ? i : -1)).filter((i) => i >= 0));
  const numCol = cols.map((_, i) => body.some((r) => isNum(r[i])) && body.every((r) => r[i] == null || isNum(r[i])));
  const pctCol = new Set(cols.map((c, i) => (c === "probability" && body.every((r) => r[i] == null || (isNum(r[i]) && r[i] >= 0 && r[i] <= 1)) ? i : -1)).filter((i) => i >= 0));
  const cell = (v, i, min, tip) => {
    if (Array.isArray(v) || (v && typeof v === "object")) return `<td class="obj">${esc(JSON.stringify(v).slice(0, 160))}</td>`;
    if (badgeCol.has(i)) return `<td>${badge(v)}</td>`;
    if (pctCol.has(i) && isNum(v)) return `<td class="num" data-v="${v}">${fmt(Math.round(v * 1000) / 10)}%</td>`;
    if (isNum(v)) return `<td class="num" data-v="${v}">${fmt(v, cols[i] === "decimals" ? 0 : min)}</td>`;
    const s = String(v ?? "");
    if (cols[i] === "measure" && /^[a-z0-9]+(_[a-z0-9]+)+$/.test(s)) return `<td>${words(s)}</td>`;
    if (isUrl(s)) return `<td>${link(s, "source")}</td>`;
    if (s.length > 48) return `<td class="text" title="${esc(tip ?? s)}"><div class="clamp">${esc(s.length > 600 ? `${s.slice(0, 597)}…` : s)}</div></td>`;
    return `<td${tip ? ` title="${esc(tip)}"` : ""}>${esc(s)}</td>`;
  };
  const head = cols.length ? `<thead><tr>${cols.map((c, i) => `<th${numCol[i] ? " class=\"num\"" : ""}>${esc(c)}</th>`).join("")}</tr></thead>` : "";
  const isHeading = (row) => row.length > 1 && row.slice(1).every((v) => v === null || v === "");
  const sortable = body.length >= 3 && !concepts && !body.some(isHeading);
  const lines = body.map((row, n) => {
    const heading = isHeading(row);
    const min = concepts && row.some((v) => isNum(v) && !Number.isInteger(v)) ? 2 : 0;
    return `<tr${heading ? " class=\"heading\"" : ""}>${row.map((v, i) => cell(v, i, min, i === 0 && concepts?.[n] ? String(concepts[n]) : undefined)).join("")}</tr>`;
  }).join("");
  const more = Array.isArray(rows) && rows.length > maxRows ? `<p class="note">${rows.length - maxRows} more rows not shown.</p>` : "";
  const filter = sortable && body.length > 12 ? `<input type="search" class="filter" placeholder="Filter ${body.length} rows" aria-label="Filter rows">` : "";
  return `<section>${title ? `<h3>${esc(title)}</h3>` : ""}${filter}<div class="scroll"><table${sortable ? ' class="sortable"' : ""}>${head}<tbody>${lines}</tbody></table></div>${note ? `<p class="note">${esc(note)}</p>` : ""}${more}</section>`;
}

const unitOf = (unit) => (unit === "%" ? "%" : unit ? `<span class="sub line">${esc(unit)}</span>` : "");

function releaseKind(columns) {
  const has = (...ks) => ks.every((k) => columns.includes(k));
  return has("guided_low", "guided_high", "actual") ? "prior" : has("measure", "low", "high", "quote") ? "guidance" : has("measure", "value", "unit", "quote") ? "results" : null;

function releaseTable({ columns, rows }, title) {

  });
  const keep = heads.map((_, i) => i < 2 || cells.some((c) => !/^<td[^>]*><\/td>$/.test(c[i])));

}

function seriesFromTable({ columns, rows }) {

  const dates = rows.map((r) => r[at]);
  const mid = (vs) => { const a = vs.filter(isNum).map(Math.abs).sort((x, y) => x - y); return a[a.length >> 1] ?? 0; };
  const pct = columns.map((c, i) => (/^percentile_/.test(String(c)) ? i : -1)).filter((i) => i > 0);
  const band = columns.includes("target_from") && columns.includes("target_to");
  let lines = [];
  for (let i = 1; i < columns.length; i++) {
    if (i === at) continue;
    if (pct.includes(i) && (band || (pct.length > 2 && i !== pct[0] && i !== pct.at(-1)))) continue;
    const values = rows.map((r) => r[i]);
    if (values.filter(isNum).length >= Math.max(3, rows.length * 0.6)) lines.push({ name: String(columns[i]), values: values.map((v) => (isNum(v) ? v : null)) });
  }
  const m0 = lines.map((l) => mid(l.values)).find((m) => m > 0);
  if (m0) lines = lines.filter((l) => { const m = mid(l.values); return m > m0 / 10 && m < m0 * 10; });
  lines = lines.slice(0, 4);
  return lines.length ? { dates, lines, ...(columns.includes("event") ? { step: true } : {}) } : null;
}

function chart({ dates, lines, step = false }, title) {
  const order = dates.map((d, i) => [d, i]).sort((a, b) => (a[0] < b[0] ? -1 : 1)).map(([, i]) => i);
  const xs = order.map((i) => dates[i]);
  const all = lines.flatMap((l) => l.values.filter(isNum));
  if (all.length < 2) return "";
  let lo = Math.min(...all), hi = Math.max(...all);
  if (lo === hi) { lo -= 1; hi += 1; }
  const W = 640, H = 220, L = 56, R = 12, T = 12, B = 28;
  const t = xs.map((d) => Date.parse(d)), t0 = t[0], span = t[t.length - 1] - t0;
  const x = (k) => L + (xs.length === 1 ? 0 : (step && span > 0 ? (t[k] - t0) / span : k / (xs.length - 1)) * (W - L - R));
  const y = (v) => T + (1 - (v - lo) / (hi - lo)) * (H - T - B);
  const grid = [0, 1, 2, 3].map((g) => { const v = lo + ((hi - lo) * g) / 3; return `<line x1="${L}" x2="${W - R}" y1="${y(v).toFixed(1)}" y2="${y(v).toFixed(1)}" class="grid"/><text x="${L - 6}" y="${(y(v) + 4).toFixed(1)}" class="axis" text-anchor="end">${short(v)}</text>`; }).join("");
  const paths = lines.map((l, n) => {
    let d = "", pen = false;
    order.forEach((i, k) => { const v = l.values[i]; if (!isNum(v)) { pen = false; return; } d += pen ? (step ? `H${x(k).toFixed(1)}V${y(v).toFixed(1)}` : `L${x(k).toFixed(1)},${y(v).toFixed(1)}`) : `M${x(k).toFixed(1)},${y(v).toFixed(1)}`; pen = true; });
    return `<path d="${d}" class="line s${n}"/>`;
  }).join("");
  const labels = `<text x="${L}" y="${H - 8}" class="axis">${esc(xs[0])}</text><text x="${W - R}" y="${H - 8}" class="axis" text-anchor="end">${esc(xs[xs.length - 1])}</text>`;
  const legend = lines.length > 1 ? `<div class="legend">${lines.map((l, n) => `<span><i class="s${n}"></i>${esc(l.name)}</span>`).join("")}</div>` : "";
  return `<section>${title ? `<h3>${esc(title)}</h3>` : ""}<svg viewBox="0 0 ${W} ${H}" role="img" aria-label="${esc(title ?? "chart")}">${grid}${paths}${labels}</svg>${legend}</section>`;
}

function ranges({ columns, rows }, title) {
  const at = (...names) => names.map((k) => columns.indexOf(k)).find((i) => i >= 0) ?? -1;
  const [m, b, lo, hi, act, u, pos] = [at("measure"), at("basis"), at("guided_low", "low"), at("guided_high", "high"), at("actual"), at("unit"), at("position")];
  if (act < 0 || lo < 0 || hi < 0) return "";
  const items = rows.filter((r) => isNum(r[lo]) && isNum(r[hi]) && isNum(r[act])).slice(0, 24);
  if (!items.length) return "";
  const html = items.map((r) => {
    const a = Math.min(r[lo], r[act]), z = Math.max(r[hi], r[act]);
    const pad = (z - a) * 0.25 || Math.abs(z) * 0.05 || 1;
    const x = (v) => ((v - (a - pad)) / (z - a + 2 * pad)) * 100;
    return `<div class="range"><div class="top"><span>${words(r[m])}${r[b] ? ` <span class="sub">${esc(r[b])}</span>` : ""}</span><span class="vals">guided ${fmt(r[lo])}${r[hi] !== r[lo] ? ` to ${fmt(r[hi])}` : ""}, actual ${fmt(r[act])}${unitAfter(r[u])}${pos >= 0 && r[pos] ? ` ${badge(r[pos])}` : ""}</span></div><div class="track"><span class="bar" style="left:${x(r[lo]).toFixed(2)}%;width:${Math.max(0.8, x(r[hi]) - x(r[lo])).toFixed(2)}%"></span><span class="dot" style="left:${x(r[act]).toFixed(2)}%"></span></div></div>`;
  }).join("");
  return `<section>${title ? `<h3>${esc(title)}</h3>` : ""}<div class="ranges">${html}</div></section>`;
}

const isObj = (v) => v !== null && typeof v === "object" && !Array.isArray(v);

function value(v, depth = 0) {
  if (isNum(v)) return fmt(v);
  if (typeof v === "boolean") return v ? "yes" : "no";
  if (typeof v === "string") return isUrl(v) ? link(v, esc(linkText(v)).replace(/\//g, "/<wbr>")) : esc(v.length > 600 ? `${v.slice(0, 599)}…` : v);
  if (Array.isArray(v) && v.every(isPlain)) return `${v.slice(0, 12).map((x) => value(x, depth + 1)).join(v.every(isUrl) ? "<br>" : ", ")}${v.length > 12 ? `, and ${v.length - 12} more` : ""}`;
  if (isObj(v) && depth === 0 && Object.values(v).every(isPlain)) return Object.entries(v).map(([k, x]) => `${words(k)} ${value(x, 1)}`).join(" · ");
  return esc(JSON.stringify(v ?? "").slice(0, 160));
}

function tablesOf(v, title) {
  if (!isObj(v) || !Array.isArray(v.columns)) return [];
  const n = v.columns.length, out = [], note = typeof v.note === "string" ? v.note : null;
  if (Array.isArray(v.values) && v.values.length === n && v.values.every(isPlain)) out.push(table({ columns: v.columns, rows: [v.values], title, note }));
  for (const [k, rows] of Object.entries(v)) {
    if (k === "rows" ? Array.isArray(rows) : k !== "columns" && Array.isArray(rows) && rows.length && rows.every((r) => Array.isArray(r) && r.length === n)) out.push(table({ columns: v.columns, rows, title: k === "rows" ? title : `${title} · ${k.replace(/_/g, " ")}`, note: k === "rows" ? note : null }));
  }
  return out;
}

const TENOR = /^\d+(\.\d+)?\s*(mo|month|months|yr|year|years|y|m|w|wk|week|weeks)$/i;
function curve(entries, title) {
  const vs = entries.map(([, v]) => v), W = 640, H = 200, L = 44, R = 12, T = 12, B = 28;
  let lo = Math.min(...vs), hi = Math.max(...vs);
  if (lo === hi) { lo -= 1; hi += 1; }
  const x = (k) => L + (k / (entries.length - 1)) * (W - L - R), y = (v) => T + (1 - (v - lo) / (hi - lo)) * (H - T - B);
  const grid = [0, 1, 2, 3].map((g) => { const v = lo + ((hi - lo) * g) / 3; return `<line x1="${L}" x2="${W - R}" y1="${y(v).toFixed(1)}" y2="${y(v).toFixed(1)}" class="grid"/><text x="${L - 6}" y="${(y(v) + 4).toFixed(1)}" class="axis" text-anchor="end">${short(v)}</text>`; }).join("");
  const path = entries.map(([, v], k) => `${k ? "L" : "M"}${x(k).toFixed(1)},${y(v).toFixed(1)}`).join("");
  const dots = entries.map(([n, v], k) => `<circle cx="${x(k).toFixed(1)}" cy="${y(v).toFixed(1)}" r="3" class="pt"><title>${esc(n)}: ${fmt(v)}</title></circle>`).join("");
  const ticks = entries.map(([n], k) => `<text x="${x(k).toFixed(1)}" y="${H - 8}" class="axis" text-anchor="middle">${esc(String(n).replace(/\s*(month|months|mo)$/i, "m").replace(/\s*(year|years|yr)$/i, "y"))}</text>`).join("");
  return `<section>${title ? `<h3>${esc(title)}</h3>` : ""}<svg viewBox="0 0 ${W} ${H}" role="img" aria-label="${esc(title ?? "curve")}">${grid}<path d="${path}" class="line s0"/>${dots}${ticks}</svg></section>`;
}

function grid(entries, title) {
  if (!entries.length) return "";
  if (entries.length >= 4 && entries.every(([k, v]) => TENOR.test(k) && isNum(v))) return curve(entries, title);
  const dd = (v) => {
    if (typeof v !== "string" || isUrl(v)) return value(v);
    if (v.length > 240) return `<div class="clamp" title="${esc(v.slice(0, 2000))}">${value(v)}</div>`;
    if (v.length > 32 && !/\s/.test(v)) return `<span title="${esc(v)}">${esc(v.slice(0, 16))}…</span>`;
    return value(v);
  };
  const wide = (v) => (Array.isArray(v) ? v.some(isUrl) : typeof v === "string" && (isUrl(v) ? linkText(v).length > 30 : v.length > 80 && /\s/.test(v)));
  return `<section>${title ? `<h3>${esc(title)}</h3>` : ""}<dl class="facts">${entries.map(([k, v]) => `<div${wide(v) ? " class=\"wide\"" : ""}><dt>${words(k)}</dt><dd>${dd(v)}</dd></div>`).join("")}</dl></section>`;
}

function facts(obj, title, skip = []) {
  if (Array.isArray(obj.columns)) {
    const plain = Object.entries(obj).filter(([k, v]) => k !== "note" && isPlain(v) && !skip.includes(k));
    return grid(plain, title) + tablesOf(obj, title).join("");
  }
  const entries = [], tables = [];
  for (const [k, v] of Object.entries(obj)) {
    if (skip.includes(k)) continue;
    const t = tablesOf(v, `${title} · ${k.replace(/_/g, " ")}`);
    if (t.length) tables.push(...t);
    else if (isObj(v) && !Object.values(v).every(isPlain)) tables.push(facts(v, `${title} · ${k.replace(/_/g, " ")}`));
    else entries.push([k, v]);
  }
  return grid(entries, title) + tables.join("");
}

function renderResult(result) {

  if (Array.isArray(result)) return result.every(isObj) ? records(result) : `<pre>${esc(JSON.stringify(result, null, 1).slice(0, 4000))}</pre>`;

  const studio = studioFor(result);

  if (studio.html) out.push(studio.html);

    } else out.push(facts(sum, "Summary"));
    seen.add("summary");
  }
  for (const [k, v] of Object.entries(result)) {
    if (seen.has(k) || STATEMENTS.includes(k) || !isObj(v) || Array.isArray(v.rows) || (Array.isArray(v.columns) && Array.isArray(v.values) && Object.keys(v).length <= 3)) continue;
    out.push(facts(v, k.replace(/_/g, " "), k === "release" ? ["headline"] : []));
    seen.add(k);
  }
  for (const k of STATEMENTS) if (!seen.has(k) && result[k]?.columns && result[k]?.rows) {
    const st = result[k];
    out.push(table({ columns: st.columns, rows: st.rows, title: `${st.title || NAMES[k]}${st.units ? ` (${st.units})` : ""}` }));
    seen.add(k);
  }
  if (Array.isArray(result.dates) && (Array.isArray(result.close) || Array.isArray(result.values)) && !seen.has("close")) {
    const values = result.close ?? result.values;
    out.push(chart({ dates: result.dates, lines: [{ name: result.close ? "close" : String(result.series ?? "value"), values }] }, result.close ? `${result.symbol ?? ""} close` : String(result.title ?? result.series ?? "")));
    seen.add("dates"); seen.add("close"); seen.add("values");
  }
  const prior = result.previous_guidance_vs_results;
  if (prior?.columns && prior?.rows) out.push(ranges(prior, "Guided and actual"));
  const tables = [];
  if (Array.isArray(result.columns) && Array.isArray(result.rows) && !seen.has("rows")) tables.push([null, result]);
  for (const [k, v] of Object.entries(result)) {
    if (seen.has(k) || !isObj(v)) continue;
    if (Array.isArray(v.columns) && (Array.isArray(v.rows) || Array.isArray(v.values))) { tables.push([k.replace(/_/g, " "), v]); seen.add(k); }
  }
  for (const [title, t] of tables) {
    const note = typeof t.note === "string" ? t.note : null;
    if (t === result && note) seen.add("note");
    if (!Array.isArray(t.rows)) { out.push(...tablesOf(t, title)); continue; }
    if (t.columns.includes("claim") && t.columns.includes("verdict")) {
      out.push(claims(t, title ?? "Claims"));
      if (note) out.push(`<p class="note">${esc(note)}</p>`);

      if (note) out.push(`<p class="note">${esc(note)}</p>`);
      continue;
    }

    const last = series ? series.dates.reduce((b, d, i) => (d > series.dates[b] ? i : b), 0) : -1;
    if (series?.step && typeof result.end === "string" && ISO.test(result.end) && result.end > series.dates[last]) {
      series.dates.push(result.end);
      for (const l of series.lines) l.values.push(l.values[last]);
    }
    if (series) out.push(chart(series, title));
    out.push(table({ columns: t.columns, rows: t.rows, title, note }));
  }
  for (const [k, v] of Object.entries(result)) {
    if (!seen.has(k) && Array.isArray(v) && v.length && v.every(isObj)) { out.push(records(v, k.replace(/_/g, " "))); seen.add(k); }
  }
  const notes = Object.entries(result).filter(([k, v]) => !seen.has(k) && /(^|_)note$/.test(k) && typeof v === "string");
  for (const [k] of notes) seen.add(k);
  const plain = Object.entries(result).filter(([k, v]) => !seen.has(k) && !["columns", "rows"].includes(k) && (isPlain(v) || (Array.isArray(v) && v.length && v.length <= 12 && v.every(isPlain)))).slice(0, 14);

  if (Array.isArray(result.limits)) out.push(`<p class="note">${result.limits.map(esc).join(" ")}</p>`);
  return out.join("") || `<pre>${esc(JSON.stringify(result, null, 1).slice(0, 4000))}</pre>`;
}

const finite = (v) => typeof v === "number" && Number.isFinite(v);
const colsOf = (t) => (t && typeof t === "object" && Array.isArray(t.columns) && Array.isArray(t.rows) ? t.columns.map(String) : null);
const pct = (v, d = 1) => (finite(v) ? `${(v * 100).toFixed(d)}%` : "–");
const signed = (v, d = 1) => (finite(v) ? `${v > 0 ? "+" : ""}${(v * 100).toFixed(d)}%` : "–");
const clamp = (x, lo, hi) => Math.max(lo, Math.min(hi, x));
const nullsOf = (n) => new Array(n).fill(null);

function smaSeries(x, n) {
  const out = nullsOf(x.length);
  if (x.length < n) return out;
  let s = 0;
  for (let i = 0; i < n; i++) s += x[i];
  out[n - 1] = s / n;
  for (let i = n; i < x.length; i++) { s += x[i] - x[i - n]; out[i] = s / n; }
  return out;
}
function emaSeries(x, n, first = null) {
  const out = nullsOf(x.length), k = 2 / (n + 1);
  let start = 0;
  while (start < x.length && x[start] === null) start++;
  const f = first ?? start + n - 1;
  if (f >= x.length) return out;
  let s = 0;
  for (let i = f - n + 1; i <= f; i++) s += x[i];
  let e = s / n;
  out[f] = e;
  for (let i = f + 1; i < x.length; i++) { e = (x[i] - e) * k + e; out[i] = e; }
  return out;
}
function rsiSeries(c, n = 14) {
  const out = nullsOf(c.length);
  if (c.length <= n) return out;
  let g = 0, l = 0;
  for (let i = 1; i <= n; i++) { const d = c[i] - c[i - 1]; if (d > 0) g += d; else l -= d; }
  g /= n; l /= n;
  out[n] = g + l === 0 ? 0 : 100 * g / (g + l);
  for (let i = n + 1; i < c.length; i++) {
    const d = c[i] - c[i - 1];
    g = (g * (n - 1) + (d > 0 ? d : 0)) / n; l = (l * (n - 1) + (d < 0 ? -d : 0)) / n;
    out[i] = g + l === 0 ? 0 : 100 * g / (g + l);
  }
  return out;
}
function macdSeries(c, fast = 12, slow = 26, signal = 9) {
  const ef = emaSeries(c, fast, slow - 1), es = emaSeries(c, slow, slow - 1);
  const m = ef.map((v, i) => (v === null || es[i] === null ? null : v - es[i])), sig = emaSeries(m, signal);
  const first = slow - 1 + signal - 1, macd = m.map((v, i) => (i < first ? null : v));
  return { macd, signal: sig, histogram: macd.map((v, i) => (v === null || sig[i] === null ? null : v - sig[i])) };
}
function bollingerSeries(c, n = 20, k = 2) {
  const mid = smaSeries(c, n), up = nullsOf(c.length), lo = nullsOf(c.length);
  for (let i = n - 1; i < c.length; i++) {
    let v = 0;
    for (let j = i - n + 1; j <= i; j++) v += (c[j] - mid[i]) ** 2;
    const sd = Math.sqrt(v / n); up[i] = mid[i] + k * sd; lo[i] = mid[i] - k * sd;
  }
  return { mid, up, lo };
}

function squarify(items, x, y, w, h) {
  const total = items.reduce((s, it) => s + it.value, 0);
  if (!(total > 0)) return [];
  const scale = (w * h) / total, queue = items.map((it) => ({ ...it, area: it.value * scale })).sort((a, b) => b.area - a.area), out = [];
  let rx = x, ry = y, rw = w, rh = h;
  const worst = (row, side) => { const s = row.reduce((a, r) => a + r.area, 0), mx = Math.max(...row.map((r) => r.area)), mn = Math.min(...row.map((r) => r.area)); return Math.max((side * side * mx) / (s * s), (s * s) / (side * side * mn)); };
  while (queue.length) {
    const side = Math.min(rw, rh), row = [queue.shift()];
    while (queue.length && worst([...row, queue[0]], side) <= worst(row, side)) row.push(queue.shift());
    const s = row.reduce((a, r) => a + r.area, 0);
    if (rw >= rh) {
      const cw = s / rh; let cy = ry;
      for (const r of row) { const ch = r.area / cw; out.push({ ...r, x: rx, y: cy, w: cw, h: ch }); cy += ch; }
      rx += cw; rw -= cw;
    } else {
      const ch = s / rw; let cx = rx;
      for (const r of row) { const cw = r.area / ch; out.push({ ...r, x: cx, y: ry, w: cw, h: ch }); cx += cw; }
      ry += ch; rh -= ch;
    }
  }
  return out;
}

function pivotChain(columns, rows) {
  const ix = Object.fromEntries(columns.map((c, i) => [c, i])), by = new Map();
  for (const r of rows) {
    const k = r[ix.strike];
    if (!finite(k)) continue;
    if (!by.has(k)) by.set(k, { strike: k, call: null, put: null });
    by.get(k)[/^p/i.test(String(r[ix.type])) ? "put" : "call"] = Object.fromEntries(columns.map((c, i) => [c, r[i]]));
  }
  return [...by.values()].sort((a, b) => a.strike - b.strike);
}

const LABEL = /^(symbol|ticker|name|asset|holding|issuer|title|company|fund)$/i;
const SIZE = /^(market_value|value|value_usd|weight|target_weight|portfolio_weight|percent|pct|percent_of_portfolio|share_of_portfolio|weight_pct)$/i;
const CHANGE = /^(change|change_pct|return|day_change|total_return|unrealized_plpc)$/i;
const GROUP = /^(sector|industry|asset_class|group|category)$/i;
function treemapData(t) {
  const c = colsOf(t);
  if (!c || t.rows.length < 2 || t.rows.length > 400) return null;
  const li = c.findIndex((x) => LABEL.test(x)), si = c.findIndex((x) => SIZE.test(x));
  if (li < 0 || si < 0) return null;
  const pl = c.indexOf("unrealized_pl"), ci = c.findIndex((x) => CHANGE.test(x)), gi = c.findIndex((x) => GROUP.test(x));
  const items = t.rows.filter((r) => finite(r[si]) && r[si] !== 0).map((r) => {
    const size = r[si], change = ci >= 0 && finite(r[ci]) ? r[ci] : pl >= 0 && finite(r[pl]) && Math.abs(size - r[pl]) > 0 ? r[pl] / Math.abs(size - r[pl]) : null;
    return { label: String(r[li] ?? ""), value: Math.abs(size), short: size < 0, size, change, group: gi >= 0 ? r[gi] : null };
  });
  if (items.filter((x) => x.value > 0).length < 2) return null;
  return { items, size: c[si], change: ci >= 0 ? c[ci] : pl >= 0 ? "return on cost" : null };
}
function weightLists(r) {
  const w = r.weights, names = r.assets ?? r.names ?? r.tickers;
  if (Array.isArray(w) && w.length >= 2 && w.length <= 400 && w.every(finite)) return { items: w.map((v, i) => ({ label: String(Array.isArray(names) ? names[i] ?? `asset ${i + 1}` : `asset ${i + 1}`), value: Math.abs(v), short: v < 0, size: v, change: null })), size: "weight", change: null };
  if (w && typeof w === "object" && !Array.isArray(w)) { const e = Object.entries(w).filter(([, v]) => finite(v)); if (e.length >= 2 && e.length <= 400) return { items: e.map(([k, v]) => ({ label: k, value: Math.abs(v), short: v < 0, size: v, change: null })), size: "weight", change: null }; }
  return null;
}

function studioFor(result) {
  const used = new Set(), out = [];
  if (!result || typeof result !== "object" || Array.isArray(result)) return { html: "", used };
  const use = (...k) => k.forEach((x) => used.add(x));
  const c = colsOf(result), n = Array.isArray(result.dates) ? result.dates.length : 0;
  if (n >= 2 && ["open", "high", "low", "close"].every((k) => Array.isArray(result[k]) && result[k].length === n)) { if (!STUDIO.compact) out.push(place("candles", result)); use("dates", "open", "high", "low", "close", "volume"); }
  if (c && result.underlying && ["strike", "type", "bid", "ask"].every((k) => c.includes(k))) { out.push(place("chain", result)); use("columns", "rows", "expirations", "underlying", "summary", "expiration", "days_to_expiration"); }
  const cc = colsOf(result.curve);
  if (cc && cc.includes("strategy") && cc.includes("buy_and_hold")) { out.push(place("backtest", result)); use("curve", "trades", "parameters", "rule", "total_return", "cagr", "volatility", "sharpe", "max_drawdown", "calmar", "buy_and_hold", "returns_tail", "periods"); }
  if (c && c[0] === "params" && typeof result.recipe === "string" && result.rows.length > 1) { out.push(place("sweep", result)); if (sweepGrid(result.rows)) use("deflation"); }
  if (result.paper === true && "token" in result && c && c.includes("accepted")) { out.push(place("ticket", result)); use("columns", "rows", "token", "expires_at", "accepted", "next"); }
  out.push(...atlasFor(result, use, used));
  if (!used.has("rows")) {
    const tm = [[null, result], ...Object.entries(result).filter(([k]) => !used.has(k))].map(([k, t]) => [k, treemapData(t)]).find(([, d]) => d) ?? [null, weightLists(result)];
    if (tm[1]) out.push(place("treemap", { ...tm[1], title: tm[0] }));
  }
  return { html: out.join(""), used };
}

const MONTHS = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
function dateLabel(s, long) {
  const m = /^(\d{4})-(\d{2})-(\d{2})/.exec(String(s));
  if (!m) return String(s);
  return long ? `${MONTHS[+m[2] - 1]} ${m[1]}` : `${MONTHS[+m[2] - 1]} ${+m[3]}`;
}
function xTicks(labels, from, to, X, y, width) {
  const count = to - from + 1, k = Math.max(2, Math.floor(width / 110)), step = Math.max(1, Math.round(count / k));
  const long = /^\d{4}/.test(String(labels[from])) && Date.parse(labels[to]) - Date.parse(labels[from]) > 400 * 86400000;
  let s = "", prev = null;
  for (let i = from; i <= to; i += step) {
    const text = dateLabel(labels[i], long);
    if (text === prev) continue;
    prev = text;
    s += `<text x="${X(i).toFixed(1)}" y="${y}" class="axis" text-anchor="${i === from ? "start" : "middle"}">${esc(text)}</text>`;
  }
  return s;
}
function yGrid(lo, hi, Y, x1, x2, fmtY = short, n = 4) {

  return s;
}
function resultOf(res) {
  if (!res) throw new Error("No answer from the host.");
  const text = res.content && res.content[0] && res.content[0].type === "text" ? res.content[0].text : "";
  if (res.isError) throw new Error(text || "The tool returned an error.");
  if (res.structuredContent !== undefined) return res.structuredContent;
  try { return JSON.parse(text); } catch { return text; }
}
function againInput(p, patch, name) {
  const inp = p.input || {};
  if (Array.isArray(inp.calls) && p.call && p.call.index !== null && p.call.index !== undefined) {
    const calls = inp.calls.slice(0, p.call.index + 1).map((c, i) => (i === p.call.index ? { name: name ?? c.name, arguments: { ...(c.arguments || {}), ...patch } } : c));
    return { calls, return: "last", digits: 6, view: true };
  }
  return { name: name ?? inp.name ?? (p.call && p.call.name), arguments: { ...(inp.arguments || {}), ...patch }, digits: 6, view: true };
}
async function callAgain(host, p, patch, name) {
  const r = resultOf(await host.call("run_tool", againInput(p, patch, name)));
  if (r && Array.isArray(r.results)) { const last = r.results.at(-1); if (last.error) throw new Error(last.error); return last.result; }
  return r;
}

  });
  if (typeof ResizeObserver === "function") { let w = 0; new ResizeObserver(() => { const now = plot.clientWidth; if (Math.abs(now - w) > 4) { w = now; draw(); } }).observe(plot); }
  draw();
}

    }
  });
  draw();
}

function mountLines(box, spec) {
  const { labels, series, height = 220, fmtY = short, area = false, markers = [], zero = false } = spec;
  box.style.position = "relative";
  box.innerHTML = '<div class="plot"></div><div class="xh" hidden></div><div class="tip" hidden></div>';
  const plot = box.querySelector(".plot"), tip = box.querySelector(".tip"), xh = box.querySelector(".xh");
  let lay = null;
  const draw = () => {
    const W = clamp(plot.clientWidth || 700, 300, 1400), L = 6, R = 62, T = 8, B = 22, n = labels.length, X = (i) => L + (n === 1 ? 0 : (i / (n - 1)) * (W - L - R));
    const all = series.flatMap((s) => s.values.filter(finite));
    let lo = Math.min(...all), hi = Math.max(...all);
    if (area) hi = Math.max(hi, 0);
    if (lo === hi) { lo -= 1; hi += 1; }
    const Y = (y) => T + (1 - (y - lo) / (hi - lo)) * (height - T - B);
    let s = yGrid(lo, hi, Y, L, W - R, fmtY);
    if (zero && lo < 0 && hi > 0) s += `<line x1="${L}" x2="${W - R}" y1="${Y(0).toFixed(1)}" y2="${Y(0).toFixed(1)}" class="guide"/>`;
    series.forEach((sr, k) => {
      const dd = pathOf(sr.values, X, Y);
      s += area ? `<path d="${dd}L${X(n - 1).toFixed(1)},${Y(0).toFixed(1)}L${X(0).toFixed(1)},${Y(0).toFixed(1)}Z" class="dd"/>` : `<path d="${dd}" class="line ${sr.cls ?? `s${k}`}"/>`;
    });
    for (const m of markers) if (finite(m.i) && finite(series[0].values[m.i])) { const x = X(m.i), y = Y(series[0].values[m.i]); s += m.side === "buy" ? `<path d="M${x.toFixed(1)},${(y + 4).toFixed(1)}l4,7h-8z" class="mk-buy"/>` : `<path d="M${x.toFixed(1)},${(y - 4).toFixed(1)}l4,-7h-8z" class="mk-sell"/>`; }
    s += xTicks(labels, 0, n - 1, X, height - 6, W - L - R);
    plot.innerHTML = `<svg viewBox="0 0 ${W} ${height}" width="${W}" height="${height}" role="img" aria-label="${esc(spec.title ?? "chart")}">${s}</svg>`;
    lay = { W, L, R, n, X };
  };
  plot.addEventListener("pointermove", (e) => {
    if (!lay) return;
    const rect = plot.getBoundingClientRect(), x = (e.clientX - rect.left) * (lay.W / rect.width), i = clamp(Math.round(((x - lay.L) / (lay.W - lay.L - lay.R)) * (lay.n - 1)), 0, lay.n - 1), px = lay.X(i) * (rect.width / lay.W);
    xh.hidden = false; xh.style.left = `${px}px`; xh.style.top = "0px"; xh.style.height = `${plot.offsetHeight}px`;
    tipAt(tip, box, [`<b>${esc(labels[i])}</b>`, ...series.map((sr) => `${esc(sr.name)} ${esc(fmtY(sr.values[i]))}`)].join("<br>"), px, 40);
  });
  plot.addEventListener("pointerleave", () => { tip.hidden = true; xh.hidden = true; });
  if (typeof ResizeObserver === "function") { let w = 0; new ResizeObserver(() => { const now = plot.clientWidth; if (Math.abs(now - w) > 4) { w = now; draw(); } }).observe(plot); }
  draw();
}

function sliderRange(key, v, periods = Infinity) {
  if (Number.isInteger(v) && !/(^|_)(z|vol|volatility|target|leverage|fraction|deviations?)($|_)/.test(key)) { const lo = /slow|exit|signal/.test(key) ? 2 : 1; return { min: lo, max: Math.max(lo + 1, Math.min(Math.max(v * 3, 20), Math.floor(periods / 2) || v * 3)), step: 1 }; }
  if (v > 0 && v <= 1) return { min: 0.01, max: 1, step: 0.01 };
  return { min: Math.round((v / 4) * 100) / 100, max: Math.round(v * 3 * 100) / 100 || 1, step: Math.abs(v) >= 10 ? 1 : 0.05 };
}

function nearest(labels, x) { const s = String(x); let b = 0; for (let i = 0; i < labels.length; i++) if (labels[i] <= s) b = i; return b; }
async function sweepCall(host, p, recipe, args, grid) {
  const inp = p.input || {};
  if (Array.isArray(inp.calls) && p.call && p.call.index !== null && p.call.index !== undefined) {
    const calls = inp.calls.slice(0, p.call.index + 1).map((c, i) => (i === p.call.index ? { name: "strategy_sweep", arguments: { recipe, arguments: args, grid } } : c));

    if (last.error) throw new Error(last.error);
    return last.result;
  }
  return resultOf(await host.call("run_tool", { name: "strategy_sweep", arguments: { recipe, arguments: args, grid }, digits: 6 }));
}

function sweepGrid(rows) {

  return { a, b, xs, ys, at: (x, y) => cell.get(`${x}|${y}`) };
}

function eigenSym(m) {
  const n = m.length, a = m.map((r) => r.map((v) => (finite(v) ? v : 0)));
  for (let sweep = 0; sweep < 80; sweep++) {
    let off = 0;
    for (let i = 0; i < n; i++) for (let j = i + 1; j < n; j++) off += a[i][j] ** 2;
    if (off < 1e-20) break;
    for (let p = 0; p < n; p++) for (let q = p + 1; q < n; q++) {
      if (Math.abs(a[p][q]) < 1e-15) continue;
      const theta = (a[q][q] - a[p][p]) / (2 * a[p][q]), t = (theta >= 0 ? 1 : -1) / (Math.abs(theta) + Math.sqrt(theta * theta + 1));
      const c = 1 / Math.sqrt(t * t + 1), s = t * c;
      for (let k = 0; k < n; k++) { const x = a[k][p], y = a[k][q]; a[k][p] = c * x - s * y; a[k][q] = s * x + c * y; }
      for (let k = 0; k < n; k++) { const x = a[p][k], y = a[q][k]; a[p][k] = c * x - s * y; a[q][k] = s * x + c * y; }
    }
  }
  return a.map((r, i) => r[i]).sort((x, y) => y - x);
}
function effectiveBets(m) {
  const ev = eigenSym(m), s = ev.reduce((a, b) => a + b, 0), q = ev.reduce((a, b) => a + b * b, 0);
  return { bets: q > 0 ? (s * s) / q : null, first_share: s > 0 ? ev[0] / s : null };
}
function correlationOf(cols) {
  const n = cols.length, m = Array.from({ length: n }, () => new Array(n).fill(1));
  for (let i = 0; i < n; i++) for (let j = i + 1; j < n; j++) {
    let k = 0, sx = 0, sy = 0, sxx = 0, syy = 0, sxy = 0;
    for (let t = 0; t < cols[i].length; t++) {
      const x = cols[i][t], y = cols[j][t];
      if (!finite(x) || !finite(y)) continue;
      k++; sx += x; sy += y; sxx += x * x; syy += y * y; sxy += x * y;
    }
    const cov = sxy - (sx * sy) / k, vx = sxx - (sx * sx) / k, vy = syy - (sy * sy) / k;
    m[i][j] = m[j][i] = k > 2 && vx > 0 && vy > 0 ? clamp(cov / Math.sqrt(vx * vy), -1, 1) : null;
  }
  return m;
}
function clusterOrder(m) {
  let groups = m.map((_, i) => [i]);
  const dist = (a, b) => { let s = 0, k = 0; for (const i of a) for (const j of b) if (finite(m[i][j])) { s += 1 - m[i][j]; k++; } return k ? s / k : 2; };
  while (groups.length > 1) {
    let bi = 0, bj = 1, bd = Infinity;
    for (let i = 0; i < groups.length; i++) for (let j = i + 1; j < groups.length; j++) { const d = dist(groups[i], groups[j]); if (d < bd) { bd = d; bi = i; bj = j; } }
    groups[bi] = [...groups[bi], ...groups[bj]];
    groups.splice(bj, 1);
  }
  return groups[0] ?? [];
}
function tenorYears(h) {
  const m = /^(\d+(?:\.\d+)?)\s*(wk|weeks?|mo|months?|yr|years?)$/i.exec(String(h).trim());

  const u = m[2].toLowerCase();
  return Number(m[1]) * (u[0] === "w" ? 1 / 52 : u[0] === "m" ? 1 / 12 : 1);
}

  const r = [];
  for (let i = 1; i < c.length; i++) r.push(c[i] / c[i - 1] - 1);
  const n = r.length, mean = n ? r.reduce((a, b) => a + b, 0) / n : 0, sd = n > 1 ? Math.sqrt(r.reduce((a, b) => a + (b - mean) ** 2, 0) / (n - 1)) : 0;
  let peak = c[0], worst = 0;
  for (const x of c) { peak = Math.max(peak, x); worst = Math.min(worst, x / peak - 1); }
  const total = c.at(-1) / c[0] - 1, years = n / 252;
  return { total, cagr: years >= 0.5 ? (1 + total) ** (1 / years) - 1 : null, volatility: sd * Math.sqrt(252), sharpe: sd > 0 ? (mean / sd) * Math.sqrt(252) : null, max_drawdown: worst, returns: r };
}
const betaOf = (y, x) => {
  const n = Math.min(x.length, y.length), mx = x.reduce((a, b) => a + b, 0) / n, my = y.reduce((a, b) => a + b, 0) / n;
  let cov = 0, vx = 0;
  for (let i = 0; i < n; i++) { cov += (x[i] - mx) * (y[i] - my); vx += (x[i] - mx) ** 2; }
  return vx > 0 ? cov / vx : null;
};
function baseLine(kind, rows, off) {
  const find = (re) => rows.findIndex((r) => re.test(String(r[0]).trim()) && r.slice(off).some(finite));
  if (kind === "balance_sheet") { const i = find(/^total assets$/i); return i >= 0 ? i : null; }
  if (kind === "income" || kind === "comprehensive_income") {
    const i = find(/^(total )?(net )?(revenues?|sales)( and other income)?,?( net)?$/i);
    if (i >= 0) return i;
    const j = find(/^(?!.*cost).*(revenue|sales)/i);
    return j >= 0 ? j : null;
  }
  return null;
}

  }

  const ts = colsOf(result.term_structure), sm = colsOf(result.smile);

function compareFor(results) {
  if (!Array.isArray(results)) return "";
  const series = [], seen = new Set();
  for (const it of results) {
    const r = it && !it.error ? it.result : null;
    if (!r || !Array.isArray(r.dates) || !Array.isArray(r.close) || r.dates.length < 30 || r.close.length !== r.dates.length) continue;
    const symbol = String(r.symbol ?? `series ${series.length + 1}`);
    if (seen.has(symbol)) continue;
    seen.add(symbol); series.push({ symbol, name: r.name ?? null, dates: r.dates, close: r.close });
  }
  return series.length >= 2 ? place("compare", { series: series.slice(0, 8), more: Math.max(0, series.length - 8) }) : "";
}

function mountStatements(el, data) {
  let tab = 0, mode = "filed";
  const picked = new Map();
  const draw = () => {
    const [key, , st] = data.items[tab], cols = st.columns, off = cols[1] === "concept" ? 2 : 1, periods = cols.slice(off), rows = st.rows;
    const ends = periods.map(periodEnd), base = baseLine(key, rows, off);
    const older = periods.map((_, j) => { const e = ends[j]; let b = -1; ends.forEach((x, k) => { if (x && e && x < e && (b < 0 || x > ends[b])) b = k; }); return b; });
    const chron = periods.map((_, j) => j).sort((a, b) => String(ends[a] ?? a).localeCompare(String(ends[b] ?? b)));
    if (!picked.has(tab)) { const want = [base, rows.findIndex((x) => /^net (income|earnings)( \(loss\))?$|^net (loss|income \(loss\))$/i.test(String(x[0]).trim())), rows.findIndex((x) => /operating activities$/i.test(String(x[0]).trim()) && x.slice(off).some(finite))]; picked.set(tab, [...new Set(want.filter((i) => i !== null && i >= 0))].slice(0, 2)); if (!picked.get(tab).length) { const f = rows.findIndex((x) => x.slice(off).some(finite)); picked.set(tab, f >= 0 ? [f] : []); } }
    const sel = picked.get(tab), shown = (row, j) => {
      const v = row[off + j];
      if (!finite(v)) return null;
      if (mode === "share") { const b = base === null ? null : rows[base][off + j]; return finite(b) && b ? v / b : null; }
      if (mode === "growth") { const k = older[j], w = k >= 0 ? row[off + k] : null; return finite(w) && w ? v / Math.abs(w) - (w < 0 ? -1 : 1) : null; }
      return v;
    };
    const cellText = (v) => (!finite(v) ? "" : mode === "filed" ? fmt(v) : mode === "share" ? pct(v) : signed(v));
    const mini = (row) => {
      const vs = chron.map((j) => row[off + j]);
      if (vs.filter(finite).length < 2) return "";
      const mx = Math.max(...vs.filter(finite).map(Math.abs)) || 1, w = 9, gap = 3, H = 18, mid = vs.some((v) => v < 0) ? H / 2 : H;
      return `<svg class="mini" viewBox="0 0 ${vs.length * (w + gap)} ${H}" width="${vs.length * (w + gap)}" height="${H}" aria-hidden="true">${vs.map((v, i) => { if (!finite(v)) return ""; const h = Math.max(1, (Math.abs(v) / mx) * (mid === H ? H : H / 2)); return `<rect x="${i * (w + gap)}" y="${(v >= 0 ? mid - h : mid).toFixed(1)}" width="${w}" height="${h.toFixed(1)}" rx="1.5" class="${v >= 0 ? "b0" : "bn"}"/>`; }).join("")}</svg>`;
    };
    const body = rows.map((row, i) => {
      const heading = row.slice(off).every((v) => v === null || v === "");
      if (heading) return `<tr class="heading"><td colspan="${periods.length + 2}">${esc(row[0])}</td></tr>`;
      return `<tr data-i="${i}" class="pick${sel.includes(i) ? " sel" : ""}${i === base && mode === "share" ? " base" : ""}"><td title="${esc(off === 2 ? String(row[1]) : "")}">${esc(row[0])}</td>${periods.map((_, j) => { const v = shown(row, j); return `<td class="num">${cellText(v)}</td>`; }).join("")}<td>${mini(row)}</td></tr>`;
    }).join("");
    const can = { share: base !== null, growth: older.some((k) => k >= 0) };
    const charted = mode === "share" && sel.length > 1 ? sel.filter((i) => i !== base) : sel;
    el.innerHTML = `${data.company?.name ? `<p class="lead">${esc(data.company.name)} <span class="sub">${esc(st.units ?? "")}</span></p>` : ""}<div class="ctl" role="toolbar" aria-label="Statement controls">${data.items.length > 1 ? `<span class="grp">${data.items.map(([, label], i) => toggle(`tab-${i}`, label, i === tab)).join("")}</span>` : ""}<span class="grp">${toggle("m-filed", "As filed", mode === "filed")}${can.share ? toggle("m-share", key === "balance_sheet" ?     bars(el.querySelector(".bars"), chron.map((j) => (ends[j] ? dateLabel(ends[j], true) : periods[j])), charted.map((i, k) => ({ name: String(rows[i][0]), values: chron.map((j) => shown(rows[i], j)), cls: `b${k}` })), mode === "filed" ? short : mode === "share" ? (v) => pct(v) : (v) => signed(v));

}
function bars(box, labels, series, fmtY) {
  if (!series.length) { box.innerHTML = ""; return; }
  const W = clamp(box.clientWidth || 700, 300, 1400), H = 190, L = 6, R = 70, T = 8, B = 22, all = series.flatMap((s) => s.values.filter(finite));
  let lo = Math.min(0, ...all), hi = Math.max(0, ...all);
  if (lo === hi) hi = 1;
  const Y = (v) => T + (1 - (v - lo) / (hi - lo)) * (H - T - B), gw = (W - L - R) / labels.length, bw = Math.min(28, (gw * 0.72) / series.length);
  let s = yGrid(lo, hi, Y, L, W - R, fmtY, 3) + `<line x1="${L}" x2="${W - R}" y1="${Y(0).toFixed(1)}" y2="${Y(0).toFixed(1)}" class="guide"/>`;
  labels.forEach((lab, g) => {
    const x0 = L + g * gw + (gw - bw * series.length) / 2;
    series.forEach((sr, k) => { const v = sr.values[g]; if (!finite(v)) return; const y = Y(Math.max(v, 0)), h = Math.max(1, Math.abs(Y(v) - Y(0))); s += `<rect x="${(x0 + k * bw + 1).toFixed(1)}" y="${y.toFixed(1)}" width="${Math.max(1, bw - 2).toFixed(1)}" height="${h.toFixed(1)}" rx="2" class="${sr.cls}"><title>${esc(`${sr.name}, ${lab}: ${fmtY(v)}`)}</title></rect>`; });
    s += `<text x="${(L + g * gw + gw / 2).toFixed(1)}" y="${H - 6}" class="axis" text-anchor="middle">${esc(lab)}</text>`;
  });
  box.innerHTML = `<svg viewBox="0 0 ${W} ${H}" width="${W}" height="${H}" role="img" aria-label="${esc(series.map((x) => x.name).join(", "))} by period">${s}</svg>`;
}
