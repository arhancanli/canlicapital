records = []

def edit(rel, changes):
    p = W/rel; original = p.read_text(); current = original; hunks = []
    for before, after, scope in changes:
        assert before != after and current.count(before) == 1, (rel, scope, current.count(before))
        current = current.replace(before, after)
        hunks.append({'old': before, 'new': after, 'scope': scope})
    inverse = current
    for h in reversed(hunks):
        assert inverse.count(h['new']) == 1
        inverse = inverse.replace(h['new'], h['old'])
    assert inverse == original
    pending[rel] = current.encode()
    records.append({'path': rel, 'original_bytes': len(original.encode()), 'original_sha256': sha(original.encode()),
                    'current_bytes': len(current.encode()), 'current_sha256': sha(current.encode()), 'exact_inverse_hunks': hunks})
