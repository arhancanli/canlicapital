def read(name, parse=True):
    path = HERE / (PREFIX + name)
    fd = os.open(path, os.O_RDONLY | os.O_NOFOLLOW)
    try:
        before = os.fstat(fd)
        with os.fdopen(fd, 'rb', closefd=False) as stream:
            data = stream.read(8 * 1024 * 1024 + 1)
        after = os.fstat(fd)
        assert len(data) <= 8 * 1024 * 1024
        assert (before.st_dev, before.st_ino, before.st_size, before.st_mtime_ns, before.st_ctime_ns) == (after.st_dev, after.st_ino, after.st_size, after.st_mtime_ns, after.st_ctime_ns)
        assert len(data) == after.st_size
    finally:
        os.close(fd)
    PINS[name] = {'path': str(path), 'bytes': len(data), 'sha256': hashlib.sha256(data).hexdigest()}
    return json.loads(data) if parse else data

def save(name, obj):
    data = (json.dumps(obj, indent=2) + '\n').encode()
    path = HERE / (PREFIX + name)
    with path.open('xb') as stream:
        stream.write(data); stream.flush(); os.fsync(stream.fileno())
    assert path.read_bytes() == data
    return {'path': str(path), 'bytes': len(data), 'sha256': hashlib.sha256(data).hexdigest()}

def tap(raw):
    clean = re.sub(r'\x1b\[[0-?]*[ -/]*[@-~]', '', raw.decode('utf-8'))
    lines = [re.sub(r'^\d{4}-\d\d-\d\dT\S+Z ?', '', line) for line in clean.splitlines()]
    suites = []
    active = None
    for number, line in enumerate(lines, 1):
        if line == 'TAP version 13':
            assert active is None
            active = {'start_line': number, 'cases': [], 'counters': {}, 'plan': None}
        if active is None:
            continue
        result = re.fullmatch(r'(ok|not ok) (\d+) - (.+)', line)
        if result:
            assert result[1] == 'ok'
            active['cases'].append({'ordinal': int(result[2]), 'name': result[3], 'status': result[1], 'line': number})
        plan = re.fullmatch(r'1\.\.(\d+)', line)
        if plan:
            assert active['plan'] is None
            active['plan'] = int(plan[1])
        counter = re.fullmatch(r'# (tests|suites|pass|fail|cancelled|skipped|todo) (\d+)', line)
        if counter:
            assert counter[1] not in active['counters']
            active['counters'][counter[1]] = int(counter[2])
        duration = re.fullmatch(r'# duration_ms ([0-9.]+)', line)
        if duration:
            active['duration_ms'] = float(duration[1]); active['end_line'] = number
            n = active['plan']; assert n is not None
            assert [case['ordinal'] for case in active['cases']] == list(range(1, n + 1))
            assert active['counters']['tests'] == n and active['counters']['pass'] == n
            assert all(active['counters'][key] == 0 for key in ('fail', 'cancelled', 'skipped', 'todo'))
            assert active['duration_ms'] > 0
            suites.append(active); active = None
    assert active is None
    return suites, lines
