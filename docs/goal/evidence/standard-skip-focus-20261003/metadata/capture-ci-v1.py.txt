"""Read-only GitHub CI custody. Does not import or execute repository code."""
from pathlib import Path
import datetime
import hashlib
import json
import os
import subprocess

HERE = Path(__file__).resolve().parent
WORK = Path('/Users/arhancanli/canlicapital-standard-skip-focus-20261003')
HEAD = '76633be4f37422932f14af850d415f509ae01f8f'
REPO = 'repos/arhancanli/canlicapital/'
PREFIX = 'pr368-76633be4-'
RECORDS = []

def save(name, data):
    path = HERE / (PREFIX + name)
    fd = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600)
    try:
        with os.fdopen(fd, 'wb') as stream:
            stream.write(data)
            stream.flush()
            os.fsync(stream.fileno())
    except BaseException:
        raise
    actual = path.read_bytes()
    assert actual == data
    return {'path': str(path), 'bytes': len(data), 'sha256': hashlib.sha256(data).hexdigest()}

def get(name, endpoint, log=False):
    argv = ['gh', 'api'] + (['--allow-escape-sequences'] if log else []) + [REPO + endpoint]
    result = subprocess.run(argv, cwd=WORK, stdout=subprocess.PIPE, stderr=subprocess.PIPE, timeout=45)
    pin = save(name, result.stdout)
    row = {'argv': argv, 'returncode': result.returncode, 'stdout': pin}
    if result.stderr:
        row['stderr'] = save(name + '.stderr', result.stderr)
    RECORDS.append(row)
    if result.returncode:
        save('capture-refusal-original.json', (json.dumps({'utc': now(), 'records': RECORDS}, indent=2) + '\n').encode())
        raise RuntimeError('GitHub metadata capture refused: ' + name)
    if log:
        assert 0 < len(result.stdout) < 8 * 1024 * 1024
        return result.stdout
    return json.loads(result.stdout)

    return datetime.datetime.now(datetime.timezone.utc).isoformat()
