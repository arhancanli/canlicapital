def continuity_append(path,addition):
    fd=os.open(path,os.O_RDWR|os.O_APPEND|os.O_NOFOLLOW)
    try:
        fcntl.flock(fd,fcntl.LOCK_EX);a=os.fstat(fd);assert stat.S_ISREG(a.st_mode) and a.st_size<8*1024*1024
        os.lseek(fd,0,os.SEEK_SET);pieces=[]
        while True:
            b=os.read(fd,1024*1024)
            if not b:break
            pieces.append(b)
        before=b''.join(pieces);assert len(before)==a.st_size
        assert identity(os.fstat(fd))==identity(os.stat(path,follow_symlinks=False))
        offset=0
        while offset<len(addition):offset+=os.write(fd,addition[offset:])
        os.fsync(fd);os.lseek(fd,0,os.SEEK_SET);pieces=[]
        while True:
            b=os.read(fd,1024*1024)
            if not b:break
            pieces.append(b)
        after=b''.join(pieces);z=os.fstat(fd)
        assert after==before+addition and identity(z)==identity(os.stat(path,follow_symlinks=False))
        return {'path':str(path),'before_bytes':len(before),'before_sha256':sha(before),'append_bytes':len(addition),'append_sha256':sha(addition),'after_bytes':len(after),'after_sha256':sha(after),'entire_prior_prefix_exact':True,'O_APPEND_flock_fsync':True,'after_identity':identity(z)}
    finally:
        fcntl.flock(fd,fcntl.LOCK_UN);os.close(fd)
